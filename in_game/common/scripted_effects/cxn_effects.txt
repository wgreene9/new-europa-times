# Variables cxn_slot_1 - 16 => these store a number from 1-16 that determines the entry in slot (tile) n
# Variables cxn_solved_1 - 4 => these store a number from 1-4 that determines the category in solved order. 0 => unsolved
# Variables cxn_solved_texture_1 - 4 => these hold the texture values for the solution bars based on the category in resp. cxn_solved_n
# Variable cxn_mistakes => holds the number of mistakes left. Starts at 4 and decrements, 0 is a list game

# Opens the window & resets the game if necessary - called through sgui
cxn_open_window = {
	# Refresh the true solution
	cxn_fetch_id = yes
	# if cache isnt set, reset the game
	if = {
		limit = {
			has_variable = cxn_cached_id
		}
		# if game is not in progress, OR cached id differs from true id, reset the game
		# else do nothing
		if = {
			limit = {
				OR = {
					NOT = { var:cxn_true_id = var:cxn_cached_id }
					NOT = { var:cxn_game > 0 }
				}
			}

			cxn_reset_game = yes
		}
	} else = {
		cxn_reset_game = yes
	}
}

# Resets the game
cxn_reset_game = {
	# set cached ID to true ID
	set_variable = { name = cxn_cached_id value = var:cxn_true_id }

	# start a new game
	set_variable = { name = cxn_game value = 1 }

	# Deselect all cells
	cxn_deselect_all = yes

	# reset scoreboard and shuffle all cells
	set_variable = { name = cxn_solved_1 value = 0 }
	set_variable = { name = cxn_solved_2 value = 0 }
	set_variable = { name = cxn_solved_3 value = 0 }
	set_variable = { name = cxn_solved_4 value = 0 }
	set_variable = { name = cxn_mistakes value = 4 }

	cxn_shuffle = yes
	cxn_show = yes
}

# Press a button
#	$tile$ - button that is pressed
cxn_press_tile = {
	if = {
		limit = {
			OR = {
				var:cxn_mistakes > 0
				var:cxn_game = 2
			}
		}
		if = {
			limit = {
				var:cxn_button_$tile$_selected = flag:cxn_tile_unselected
			}
			# tile is unselected, select it
			if = {
				limit = {
					var:cxn_selected_count < 4
				}
				change_variable = { name = cxn_selected_count add = 1 }
				set_variable = { name = cxn_button_$tile$_selected value = flag:cxn_tile_selected }
			}
		} else = {
			# tile is selected, unselect it
			change_variable = { name = cxn_selected_count subtract = 1 }
			set_variable = { name = cxn_button_$tile$_selected value = flag:cxn_tile_unselected }
		}
	}
}

# Shuffle --
#	1. initialize cxn_shuffler_index to 1
#	2. while loop that excludes indices belonging to a solution
#	3. Random list - for every remaining value between cxn_var_value and 16 inclusive, set a random slot (without a solution bar in that row) to that value
cxn_shuffle = {
	# deselect all
	cxn_deselect_all = yes

	# Init index
	set_variable = { name = cxn_shuffler_index value = 1 }

	# Set slots to 0
	set_variable = { name = cxn_slot_1 value = 0 }
	set_variable = { name = cxn_slot_2 value = 0 }
	set_variable = { name = cxn_slot_3 value = 0 }
	set_variable = { name = cxn_slot_4 value = 0 }
	set_variable = { name = cxn_slot_5 value = 0 }
	set_variable = { name = cxn_slot_6 value = 0 }
	set_variable = { name = cxn_slot_7 value = 0 }
	set_variable = { name = cxn_slot_8 value = 0 }
	set_variable = { name = cxn_slot_9 value = 0 }
	set_variable = { name = cxn_slot_10 value = 0 }
	set_variable = { name = cxn_slot_11 value = 0 }
	set_variable = { name = cxn_slot_12 value = 0 }
	set_variable = { name = cxn_slot_13 value = 0 }
	set_variable = { name = cxn_slot_14 value = 0 }
	set_variable = { name = cxn_slot_15 value = 0 }
	set_variable = { name = cxn_slot_16 value = 0 }

	while = {
		limit = {
			var:cxn_shuffler_index <= 16
		}

		# If the shuffler index is 1-4 and there is a solution cat 1, 5-8 with cat 2, etc. then skip
		if = {
			limit = {
				OR = {
					# between 1-4
					AND = {
						var:cxn_shuffler_index > 0
						var:cxn_shuffler_index < 5
						OR = {
							var:cxn_solved_1 = 1
							var:cxn_solved_2 = 1
							var:cxn_solved_3 = 1
							var:cxn_solved_4 = 1
						}
					}
					# between 5-8
					AND = {
						var:cxn_shuffler_index > 4
						var:cxn_shuffler_index < 9
						OR = {
							var:cxn_solved_1 = 2
							var:cxn_solved_2 = 2
							var:cxn_solved_3 = 2
							var:cxn_solved_4 = 2
						}
					}
					# between 9-12
					AND = {
						var:cxn_shuffler_index > 8
						var:cxn_shuffler_index < 13
						OR = {
							var:cxn_solved_1 = 3
							var:cxn_solved_2 = 3
							var:cxn_solved_3 = 3
							var:cxn_solved_4 = 3
						}
					}
					# between 13-16
					AND = {
						var:cxn_shuffler_index > 12
						var:cxn_shuffler_index < 17
						OR = {
							var:cxn_solved_1 = 4
							var:cxn_solved_2 = 4
							var:cxn_solved_3 = 4
							var:cxn_solved_4 = 4
						}
					}
				}
			}
			# more readable than NOT-ting the thing above
			# and idt empty ifs work
			set_variable = { name = cxn_unused value = 1 }
		} else = {
			# This is a valid item to shuffle
			# Random list
			random_list = {
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_1 = 0
							var:cxn_slot_1 = 0
						}
					}
					set_variable = { name = cxn_slot_1 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_1 = 0
							var:cxn_slot_2 = 0
						}
					}
					set_variable = { name = cxn_slot_2 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_1 = 0
							var:cxn_slot_3 = 0
						}
					}
					set_variable = { name = cxn_slot_3 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_1 = 0
							var:cxn_slot_4 = 0
						}
					}
					set_variable = { name = cxn_slot_4 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_2 = 0
							var:cxn_slot_5 = 0
						}
					}
					set_variable = { name = cxn_slot_5 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_2 = 0
							var:cxn_slot_6 = 0
						}
					}
					set_variable = { name = cxn_slot_6 value = var:cxn_shuffler_index }
				}
				1 = {

						AND = {
							var:cxn_solved_2 = 0
							var:cxn_slot_7 = 0
						}
					}
					set_variable = { name = cxn_slot_7 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_2 = 0
							var:cxn_slot_8 = 0
						}
					}
					set_variable = { name = cxn_slot_8 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_3 = 0
							var:cxn_slot_9 = 0
						}
					}
					set_variable = { name = cxn_slot_9 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_3 = 0
							var:cxn_slot_10 = 0
						}
					}
					set_variable = { name = cxn_slot_10 value = var:cxn_shuffler_index }
				}
				1 = {

							var:cxn_slot_11 = 0
						}
					}
					set_variable = { name = cxn_slot_11 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_3 = 0
							var:cxn_slot_12 = 0
						}
					}
					set_variable = { name = cxn_slot_12 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_4 = 0
							var:cxn_slot_13 = 0
						}
					}
					set_variable = { name = cxn_slot_13 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_4 = 0
							var:cxn_slot_14 = 0
						}
					}
					set_variable = { name = cxn_slot_14 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_4 = 0
							var:cxn_slot_15 = 0
						}
					}
					set_variable = { name = cxn_slot_15 value = var:cxn_shuffler_index }
				}
				1 = {
					trigger = {
						AND = {
							var:cxn_solved_4 = 0
							var:cxn_slot_16 = 0
						}
					}
					set_variable = { name = cxn_slot_16 value = var:cxn_shuffler_index }
				}
			}
		}

		# Increment counter
		change_variable = { name = cxn_shuffler_index add = 1 }
	}
}

# Submit --

#	3. check if the 4 selected are 1234 5678 etc.
#		- if they are, save into solution var and reshuffle / reshow
#		- if they aren't, mark a mistake
cxn_submit = {
	if = {
		limit = {
			AND = {
				var:cxn_selected_count = 4
				var:cxn_mistakes > 0
			}
		}

		# 4 are selected. Identify the 4 selected and save them into variables
		# Variables - cxn_select_temp_1 - 4

		set_variable = { name = cxn_submit_counter value = 0 }
		
		# save variables with helper function

		cxn_submit_helper = { tile = 13 }

		# 4 values are saved in the temps

		if = {

			set_variable = { name = cxn_submit_solution value = 1 }

						var:cxn_select_temp_3 = 16

		# if solved successfully, deselect all & show; or else increase mistakes by 1
		if = {

			}
			cxn_shuffle = yes
			cxn_deselect_all = yes

						var:cxn_solved_4 > 0

				}
				set_variable = {
					name = cxn_game

				}
			}
		} else = {
			change_variable = { name = cxn_mistakes subtract = 1 }
			if = {
				limit = {
					var:cxn_mistakes = 0
				}
				cxn_deselect_all = yes
			}
		}
	}
}

# helper for submit
cxn_submit_helper = {
	if = {
		limit = {
			var:cxn_button_$tile$_selected = flag:cxn_tile_selected
		}
		if = {
			limit = {
				var:cxn_submit_counter = 0
			}
			set_variable = { name = cxn_select_temp_1 value = var:cxn_slot_$tile$ }
		} else_if = {
			limit = {
				var:cxn_submit_counter = 1
			}
			set_variable = { name = cxn_select_temp_2 value = var:cxn_slot_$tile$ }
		} else_if = {
			limit = {
				var:cxn_submit_counter = 2
			}
			set_variable = { name = cxn_select_temp_3 value = var:cxn_slot_$tile$ }
		} else_if = {
			limit = {
				var:cxn_submit_counter = 3
			}
			set_variable = { name = cxn_select_temp_4 value = var:cxn_slot_$tile$ }
		}
		change_variable = { name = cxn_submit_counter add = 1 }
	}
}
