# Effect called upon opening window
wordle_open_window = {
	# Refresh the true solution
	wordle_fetch_solution = yes
	# if cache isnt set, reset the game
	if = {
		limit = {
			has_variable = wordle_cached_id
		}
		# if game is not in progress, OR cached id differs from true id, reset the game
		# else do nothing
		if = {
			limit = {
				OR = {
					NOT = { var:wordle_true_id = root.var:wordle_cached_id }
					NOT = { var:wordle_game > 0 }
				}
			}

			wordle_reset_game = yes
		}
	} else = {
		wordle_reset_game = yes
	}
	
}

# press enter
# GIGA TODO, this needs win checking
wordle_press_enter = {
    # only do something if game is in progress
    if = {
        limit = {
            var:wordle_game = 1
        }
        # if col cursor is 6, check if the word is valid, and if it is, increment row cursor and set column cursor to 1 again
        if = {
            limit = {
                var:wordle_col_cursor = 6
            }
            # Check if game is won
            if = {
                limit = {
                    wordle_win = yes
                }
                # increement row cursor and reset col cursor (for the show functionality)
                change_variable = { name = wordle_row_cursor add = 1 }
                set_variable = { name = wordle_col_cursor value = 1 }

                # recompute show
                wordle_recompute_show = yes

                # set game to win
                set_variable = { name = wordle_game value = 2 }
            }
            # Check if word is valid
            else_if = {
                limit = {
                    OR = {
                        AND = {
                            var:wordle_row_cursor = 1
                            wordle_valid_guess = { r = 1 }
                        }
                        AND = {
                            var:wordle_row_cursor = 2
                            wordle_valid_guess = { r = 2 }
                        }
                        AND = {
                            var:wordle_row_cursor = 3
                            wordle_valid_guess = { r = 3 }
                        }
                        AND = {
                            var:wordle_row_cursor = 4
                            wordle_valid_guess = { r = 4 }
                        }
                        AND = {
                            var:wordle_row_cursor = 5
                            wordle_valid_guess = { r = 5 }
                        }
                        AND = {
                            var:wordle_row_cursor = 6
                            wordle_valid_guess = { r = 6 }
                        }
                    }
                }
                # increement row cursor and reset col cursor
                change_variable = { name = wordle_row_cursor add = 1 }
                set_variable = { name = wordle_col_cursor value = 1 }

                # recompute show
                wordle_recompute_show = yes
            }
        }
    }
	
}

# r, c
wordle_green_pass = {
    if = {
        limit = {
            AND = {
                wordle_letter_counter_positive = { var = wordle_cell_$r$_$c$ }
                var:wordle_cell_$r$_$c$ = var:wordle_cached_$c$
            }
        }
        wordle_show_cell = { var = wordle_cell_$r$_$c$ r = $r$ c = $c$ color = green }
        wordle_dec_letter_counter = { var = wordle_cell_$r$_$c$ }
    }
}

# r, c
wordle_yellow_pass = {
    if = {
        limit = {
            NOT = { var:wordle_cell_$r$_$c$ = var:wordle_cached_$c$ }
        }
        if = {
            limit = {
                wordle_letter_counter_positive = { var = wordle_cell_$r$_$c$ }
            }
            wordle_show_cell = { var = wordle_cell_$r$_$c$ r = $r$ c = $c$ color = yellow }
            wordle_dec_letter_counter = { var = wordle_cell_$r$_$c$ }
        } else = {
            wordle_show_cell = { var = wordle_cell_$r$_$c$ r = $r$ c = $c$ color = gray }
        }
    }
}

# show an entire row [& perform appropriate keyboard operations]
#   $r$ - row
wordle_show_row = {

    # compute the counts
    wordle_reset_letter_counters = yes
    wordle_build_letter_counters = yes

    # green pass - if letter counter is positive and letters match
    wordle_green_pass = { r = $r$ c = 1 }
    wordle_green_pass = { r = $r$ c = 2 }
    wordle_green_pass = { r = $r$ c = 3 }
    wordle_green_pass = { r = $r$ c = 4 }
    wordle_green_pass = { r = $r$ c = 5 }

    # yellow pass - sets to either yellow or gray depending on counter
    wordle_yellow_pass = { r = $r$ c = 1 }
    wordle_yellow_pass = { r = $r$ c = 2 }
    wordle_yellow_pass = { r = $r$ c = 3 }
    wordle_yellow_pass = { r = $r$ c = 4 }
    wordle_yellow_pass = { r = $r$ c = 5 }
}

# recompute show - todo ADD IN COLORS
wordle_recompute_show = {
    if = {
        limit = { var:wordle_row_cursor > 1 }
        wordle_show_row = { r = 1 }
    } else = {
        wordle_show_unfinished_row = { r = 1 }
    }
    if = {
        limit = { var:wordle_row_cursor > 2 }
        wordle_show_row = { r = 2 }
    } else = {
        wordle_show_unfinished_row = { r = 2 }
    }
    if = {
        limit = { var:wordle_row_cursor > 3 }
        wordle_show_row = { r = 3 }
    } else = {
        wordle_show_unfinished_row = { r = 3 }
    }
    if = {
        limit = { var:wordle_row_cursor > 4 }
        wordle_show_row = { r = 4 }
    } else = {
        wordle_show_unfinished_row = { r = 4 }
    }
    if = {
        limit = { var:wordle_row_cursor > 5 }
        wordle_show_row = { r = 5 }
    } else = {
        wordle_show_unfinished_row = { r = 5 }
    }
    if = {
        limit = { var:wordle_row_cursor > 6 }
        wordle_show_row = { r = 6 }
    } else = {
        wordle_show_unfinished_row = { r = 6 }
    }
