cxn_shuffle_wrapper = {
	effect = {
		cxn_shuffle = yes
		cxn_show = yes
	}
}

cxn_deselect_all_wrapper = {
	effect = {
		cxn_deselect_all = yes
	}
}

cxn_submit_wrapper = {
	effect = {
		cxn_submit = yes
	}
}

cxn_show_shuffle_button = {
	is_shown = {
        var:cxn_mistakes > 0
			var:cxn_game < 2
    }
}

cxn_deselect_all_clickable = {
    is_shown = {
        AND = {
			NOT = { var:cxn_selected_count = 0 }
			var:cxn_mistakes > 0
			var:cxn_game < 2
		}
    }
}

cxn_deselect_all_unclickable = {
    is_shown = {
		AND = {
			var:cxn_selected_count = 0
			var:cxn_mistakes > 0
			var:cxn_game < 2
		}
    }
}

cxn_submit_clickable = {
    is_shown = {
		AND = {
        	var:cxn_selected_count = 4
			var:cxn_mistakes > 0
			var:cxn_game < 2
		}
    }
}

cxn_submit_unclickable = {
    is_shown = {
        AND = {
			NOT = { var:cxn_selected_count = 4 }
			var:cxn_mistakes > 0
			var:cxn_game < 2
		}
    }
}
